import { useEffect } from 'react';

export function useBackToClose(isOpen: boolean, onClose: () => void) {
  useEffect(() => {
    if (isOpen) {
      // 1. Push a new state when the modal opens to make the back button target the modal
      window.history.pushState({ modalOpen: true }, "", window.location.href);

      const handlePopState = (event: PopStateEvent) => {
        // 2. Intercept the back button action
        event.preventDefault(); 
        onClose(); // Call the component's closing function
      };

      window.addEventListener('popstate', handlePopState);

      return () => {
        window.removeEventListener('popstate', handlePopState);
        // CRITICAL: Clean up by navigating back out of the temporary state
        if (window.history.state && window.history.state.modalOpen) {
           window.history.back();
        }
      };
    }
  }, [isOpen, onClose]);
}
