// 2. Add State inside the component
const [fullReport, setFullReport] = useState<any>(null);
const [loadingFull, setLoadingFull] = useState(false);
const [showFullModal, setShowFullModal] = useState(false);

// 3. Add Handler Function
const handleOpenFullReport = async () => {
  setLoadingFull(true);
  try {
    const res = await fetch('/api/academy/full-report');
    if (res.status === 403) {
      alert("This report is classified. Upgrade to Premium to access the full 3-page briefing.");
      setLoadingFull(false);
      return;
    }
    const json = await res.json();
    if (json.success) {
      setFullReport(json.data);
      setShowFullModal(true);
    }
  } catch (e) {
    console.error(e);
  } finally {
    setLoadingFull(false);
  }
};
