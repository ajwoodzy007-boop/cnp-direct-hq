// client/components/TheOracle.tsx
import React, { useState, useEffect } from 'react';
import { Target, ArrowRight, X, Activity, BarChart2, FileText, AlertTriangle, Loader2 } from 'lucide-react';

interface PickData {
  ticker: string;
  entryPrice: number;
  predictedPrice: number;
  outcome: string;
  confidence: string;
  signal: string;
  rsi?: number;     
  sentimentScore?: number;
  rvol?: number;
}

export default function TheOracle() {
  const [picks, setPicks] = useState<PickData[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedPick, setSelectedPick] = useState<PickData | null>(null);
  
  // NEW: State for Real AI Data
  const [aiReport, setAiReport] = useState<any>(null);
  const [analyzing, setAnalyzing] = useState(false);

  // Mock stats
  const stats = { wins: 12, losses: 4, winRate: 75, streak: 3 };

  useEffect(() => {
    async function fetchDailyPicks() {
      try {
        const res = await fetch('/api/oracle/daily');
        const json = await res.json();
        if (json.success) setPicks(json.data);
      } catch (e) { console.error(e); } finally { setLoading(false); }
    }
    fetchDailyPicks();
  }, []);

  // NEW: Function to call GPT-4
  const generateReport = async (pick: PickData) => {
    setAnalyzing(true);
    setAiReport(null); // Clear old report
    try {
      const res = await fetch('/api/ai/analyze', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ticker: pick.ticker,
          price: pick.entryPrice,
          rsi: pick.rsi || 50,
          trend: pick.signal
        })
      });
      const json = await res.json();
      if (json.success) setAiReport(json.data);
    } catch (e) {
      console.error(e);
    } finally {
      setAnalyzing(false);
    }
  };

  // Trigger AI when a pick is opened
  useEffect(() => {
    if (selectedPick) {
      generateReport(selectedPick);
    }
  }, [selectedPick]);

  return (
    <div className="space-y-8 animate-in fade-in duration-500 relative">
      
      {/* ... (Keep HERO SECTION and DAILY PICKS GRID the same as before) ... */}
      
      {/* HERO SECTION */}
      <div className="bg-gradient-to-br from-slate-900 via-slate-900 to-cyan-950 p-8 rounded-2xl border border-slate-800 relative overflow-hidden">
        <div className="relative z-10">
          <h2 className="text-3xl font-bold text-white flex items-center gap-3">
            <Target className="text-cyan-400 h-8 w-8" /> The Oracle
          </h2>
          <p className="text-slate-400 mt-2 max-w-2xl">High-conviction setups filtered by the Sentinel Engine.</p>
        </div>
      </div>

      {/* GRID */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {picks.map((pick) => (
          <div key={pick.ticker} className="bg-slate-900 border border-slate-800 rounded-xl p-6 hover:border-cyan-500/30 transition-all group">
            <h4 className="text-2xl font-bold text-white mb-4">{pick.ticker}</h4>
            <div className="space-y-3 mb-6">
               <div className="flex justify-between text-sm"><span className="text-slate-500">Entry</span><span className="text-white">${pick.entryPrice.toFixed(2)}</span></div>
               <div className="flex justify-between text-sm"><span className="text-slate-500">Signal</span><span className="text-cyan-400 font-bold">{pick.signal}</span></div>
            </div>
            <button 
              onClick={() => setSelectedPick(pick)}
              className="w-full py-2 bg-slate-800 hover:bg-cyan-600 hover:text-white text-slate-300 rounded-lg text-sm font-medium transition-all flex items-center justify-center gap-2"
            >
              View Analysis <ArrowRight className="h-4 w-4" />
            </button>
          </div>
        ))}
      </div>

      {/* --- LIVE AI MODAL --- */}
      {selectedPick && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-sm animate-in fade-in duration-200">
          <div className="bg-slate-900 border border-slate-700 w-full max-w-lg rounded-2xl shadow-2xl overflow-hidden">
            
            {/* Header */}
            <div className="p-6 border-b border-slate-800 flex justify-between items-center bg-slate-800/50">
              <div>
                <h3 className="text-xl font-bold text-white flex items-center gap-2">
                  <FileText className="text-cyan-400 h-5 w-5" />
                  Tactical Report: {selectedPick.ticker}
                </h3>
                <p className="text-xs text-slate-400 mt-1 uppercase tracking-wider flex items-center gap-2">
                   Powered by OpenAI GPT-4o
                </p>
              </div>
              <button onClick={() => setSelectedPick(null)} className="text-slate-400 hover:text-white">
                <X className="h-6 w-6" />
              </button>
            </div>

            {/* Content Area */}
            <div className="p-6 space-y-6 min-h-[300px]">
              
              {analyzing ? (
                <div className="flex flex-col items-center justify-center h-48 space-y-4">
                  <Loader2 className="h-10 w-10 text-cyan-500 animate-spin" />
                  <div className="text-sm text-cyan-400 animate-pulse">Establishing Uplink to Sentinel Core...</div>
                </div>
              ) : aiReport ? (
                <>
                  {/* 1. The Verdict */}
                  <div className={`border p-4 rounded-lg flex items-start gap-4 ${
                    aiReport.verdict.includes('BUY') 
                      ? 'bg-green-500/10 border-green-500/20' 
                      : 'bg-red-500/10 border-red-500/20'
                  }`}>
                    <AlertTriangle className={`h-6 w-6 shrink-0 mt-1 ${
                       aiReport.verdict.includes('BUY') ? 'text-green-400' : 'text-red-400'
                    }`} />
                    <div>
                      <div className={`font-bold ${
                         aiReport.verdict.includes('BUY') ? 'text-green-400' : 'text-red-400'
                      }`}>
                        AI Verdict: {aiReport.verdict}
                      </div>
                      <p className="text-sm text-slate-300 mt-2 leading-relaxed">
                        {aiReport.summary}
                      </p>
                    </div>
                  </div>

                  {/* 2. Risk Assessment */}
                  <div className="bg-slate-950 p-4 rounded-lg border border-slate-800">
                    <div className="text-xs text-slate-500 uppercase font-bold mb-1">Downside Risk</div>
                    <p className="text-sm text-slate-400 italic">"{aiReport.risk}"</p>
                  </div>

                  {/* 3. Tech Specs */}
                  <div className="grid grid-cols-2 gap-4">
                     <div className="bg-slate-950 p-3 rounded border border-slate-800 text-center">
                        <div className="text-xs text-slate-500">RSI Strength</div>
                        <div className="font-mono text-white font-bold">{selectedPick.rsi || 50}</div>
                     </div>
                     <div className="bg-slate-950 p-3 rounded border border-slate-800 text-center">
                        <div className="text-xs text-slate-500">Confidence</div>
                        <div className="font-mono text-cyan-400 font-bold">HIGH</div>
                     </div>
                  </div>
                </>
              ) : (
                <div className="text-center text-red-400">Analysis Failed. Uplink Error.</div>
              )}

            </div>

            {/* Footer */}
            <div className="p-4 bg-slate-950 border-t border-slate-800 flex gap-3">
              <button 
                onClick={() => setSelectedPick(null)}
                className="flex-1 py-2 bg-slate-800 hover:bg-slate-700 text-white rounded-lg text-sm font-medium transition-colors"
              >
                Close Report
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
